\begin{frame}[label=urls]{三类 URL 如何读出股票、主题与展示参数}

  \medskip
  \leadrow{复合图表 · 第 7 行}{\urllit{.../earnings/update\_earningsChart.asp?duration=1095}\\

\begin{frame}[label=rules]{三条规则把浏览研究连到一笔订单}
  \crumb{Part 2}{2.2 URL 识别与交易配对}{Table IA1 · Steps 1、2、8 · p.\,37}
  \small
  \leadrow{1 · 补股票代码}{\texttt{IMCL} \(\longrightarrow\) 缺代码新闻 \(\longrightarrow\) \texttt{IMCL}\\
  同会话前后最近的已匹配记录均为同一只股票时补配。}
  \medskip
  \leadrow{2 · 划会话、合并重复订单}{
   Yahoo／财经站 \(\longrightarrow\) 券商 \(\longrightarrow\) \texttt{GOOG sell}\\
   相邻网页点击间隔 \(\leq 15\) 分钟为同会话；同股、同向候选订单间隔 \(\leq 30\) 分钟合并，取末次时点。}
  \medskip
  \leadrow{3 · 归属下一笔同股交易}{
    周二研究 \(\longrightarrow\) 周三卖出 \qquad 周四研究 \(\longrightarrow\) 周五买入\\
    观察期内此后无同股交易，归入末笔后的研究。周历为规则示意。}
  \finding{得到每户、每股、带内容和时间的研究记录，以及可用于归属的订单时间。}

\end{frame}
